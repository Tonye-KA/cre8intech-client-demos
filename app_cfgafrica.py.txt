# TAB 1: CLIENT-FACING INVESTOR NAVIGATOR
# ==============================================================================
with tab1:
    st.subheader("Investor Product Matcher")
    st.write("Public onboarding tool designed for `cfgafrica.com` or WhatsApp to match prospects with CFG products.")
    
    with st.form("public_navigator_form"):
        user_type = st.selectbox(
            "1. What best describes your investor profile?",
            [
                "Working Professional / Salary Earner (Seeking inflation protection & steady growth)",
                "Diaspora / Remote Earner (Managing multi-currency income in USD / GBP / EUR)",
                "Faith-Conscious / Ethical Investor (Requiring strict Sharia / Halal compliance)",
                "Business Owner / High-Net-Worth Individual (Focusing on capital preservation & succession)",
                "Corporate / Institutional Treasurer (Managing short-to-medium liquidity)"
            ]
        )

        core_goal = st.selectbox(
            "2. What is your primary investment objective?",
            [
                "Stable monthly/quarterly passive income (Shielded from market volatility)",
                "Foreign currency hedge & wealth preservation (USD / GBP placements)",
                "Ethical & interest-free (Riba-free) capital appreciation",
                "Long-term generational wealth transfer & private trust setup",
                "Liquid corporate treasury reserve (90 - 180 days roll-over)"
            ]
        )

        capital_horizon = st.selectbox(
            "3. Preferred investment horizon & risk tolerance:",
            [
                "Short-Term Liquidity (30 - 90 Days) with Low Risk",
                "Medium-Term Growth (6 - 12 Months) with Moderate Risk",
                "Long-Term Compounding (1 - 5 Years) with Capital Preservation",
                "Generational / Structured Estate (Multi-Year Trust Mandate)"
            ]
        )

        submit_public = st.form_submit_button("Generate CFG Investment Plan 🚀")

    if submit_public:
        if not api_key:
            st.error("Please add OPENAI_API_KEY in Streamlit Secrets.")
        else:
            client = openai.OpenAI(api_key=api_key)
            system_prompt_public = """
            You are the CFG Wealth Navigator for CFG Africa. Match the user with CFG's products:
            - CFG AM Naira Fixed Income Fund (SEC-regulated mutual fund)
            - Multi-Currency Tenured Placements (USD, GBP, EUR, NGN)
            - Non-Interest / Halal Ethical Portfolios (Sharia/Ijarah-based)
            - Fixed Rate Notes & Depository Notes
            - CFG Private Wealth, Family Office & Estate Planning Trusts
            
            OUTPUT:
            1. Strategy Summary (1-2 lines)
            2. Primary Recommended CFG Product
            3. Complementary Product for Diversification
            4. Suggested Allocation %
            5. Clear CTA to complete onboarding with CFG.
            """
            prompt = f"Profile: {user_type} | Goal: {core_goal} | Horizon: {capital_horizon}"
            with st.spinner("Structuring portfolio allocation..."):
                res = client.chat.completions.create(
                    model="gpt-4o-mini",
                    messages=[{"role": "system", "content": system_prompt_public}, {"role": "user", "content": prompt}]
                )
                st.success("Investment Recommendation Generated:")
                st.markdown(res.choices[0].message.content)
